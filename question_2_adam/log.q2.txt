> ### Deriving EOSSTT ####
> # Function formatting the ESSOTT values mapped from the DS.DSDECOD values
> ## Logic: if DSDECOD -> "COMPLETED, ESSOTT -> "COMPLETED"
> ##        if DSDECOD -> "SCREEN FAILURE", ESSOTT -> NA (character)
> ##        if anything else, ESSOTT -> DISCONTINUED
> # --------------------------------------------------------------------------------
> format_eosstt <- function(x) {    # Function to sift through values, return proper ESSOTT mappings
+   case_when(
+     x %in% c("COMPLETED")      ~ "COMPLETED",
+     x %in% c("SCREEN FAILURE") ~ NA_character_,
+     TRUE                       ~ "DISCONTINUED"
+   )
+ }
